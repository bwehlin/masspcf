100    /// In-place max-shift exponentiation of a log-weight row:
101    /// w_i = exp(e_i - max_j e_j). The shift cancels on normalization, and
102    /// the best point's weight is exactly 1, so the row can never underflow
103    /// to all-zero.
104    template <typename T>
105    void log_row_to_weights(std::span<T> row)
106    {
107      T maxLog = -std::numeric_limits<T>::infinity();
108      for (T e : row)
109      {
110        maxLog = std::max(maxLog, e); // drops NaNs; exp keeps them NaN below
111      }
112      if (!(maxLog > -std::numeric_limits<T>::infinity()))
113      {
114        maxLog = T(0); // all -inf (empty region): avoid (-inf) - (-inf) = NaN
115      }
116      for (T &e : row)
117      {
118        e = std::exp(e - maxLog);
119      }
120    }
