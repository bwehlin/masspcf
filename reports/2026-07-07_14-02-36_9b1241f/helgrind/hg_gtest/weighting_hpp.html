23  // Sampling weights: how each reference point earns its draw probability.
24  //
25  // A filter maps a (query point, reference point) pair to a scalar; a
26  // distribution maps that value to a non-negative weight. Both are plain
27  // functors, so custom callables work like the built-ins.
28  //
29  // A distribution may additionally provide `T log_weight(T v) const`
30  // (= log(operator()(v)), see Gaussian): its rows are then evaluated in log
31  // space and finished with a max-shift exponentiation, so weights that
32  // mathematically never vanish cannot underflow to an all-zero row. Either
33  // way, code downstream of compute_weights only ever sees plain weights.
34  // ===========================================================================

36  /// Euclidean distance between query point @p queryIdx and reference point
37  /// @p refIdx. The samplers validate the shared dimension once up front, so
38  /// this hot-path operator stays branch-free.
39  template <typename T>
40  struct EuclideanDistance
41  {
42    T operator()(const PointCloud<T>& query, size_t queryIdx,
43                 const PointCloud<T>& reference, size_t refIdx) const
44    {
45      T acc = T(0);
46      for (size_t k = 0; k < query.dim(); ++k)
47      {
48        T d = query(queryIdx, k) - reference(refIdx, k);
49        acc += d * d;
50      }
51      return std::sqrt(acc);
52    }
53  };

55  /// Weight 1 when the filter value lies in [low, high], 0 outside. With the
56  /// distance filter that is a disk (low = 0), an annulus, or the whole cloud
57  /// (the default, high = +inf). Member names match the Python spec.
58  template <typename T>
59  struct Uniform
60  {
61    T low = T(0);
62    T high = std::numeric_limits<T>::infinity();
63
64    T operator()(T v) const { return (v >= low && v <= high) ? T(1) : T(0); }
65  };

67  /// Unnormalized Gaussian exp(-((v - mean) / sigma)^2 / 2) of the filter
68  /// value, offering the log-space channel: a raw exp would underflow to an
69  /// artificial all-zero row once every reference point sits ~38 sigma
70  /// (float64; ~14 float32) from the mean, while the max-shifted exp samples
71  /// this exact Gaussian at any distance.
72  template <typename T>
73  struct Gaussian
74  {
75    T mean = T(0);
76    T sigma = T(1);
77
78    T operator()(T v) const { return std::exp(log_weight(v)); }
79
80    T log_weight(T v) const
81    {
82      T d = (v - mean) / sigma;
83      return T(-0.5) * d * d;
84    }
85  };

91    /// returning log(operator()(v)). Plain functors are evaluated directly.

97    /// In-place max-shift exponentiation of a log-weight row:
98    /// w_i = exp(e_i - max_j e_j). The shift cancels on normalization, and
99    /// the best point's weight is exactly 1, so the row can never underflow
100    /// to all-zero.
101    template <typename T>
102    void log_row_to_weights(std::span<T> row)
103    {
104      T maxLog = -std::numeric_limits<T>::infinity();
105      for (T e : row)
106        maxLog = std::max(maxLog, e);  // drops NaNs; exp keeps them NaN below
107      if (!(maxLog > -std::numeric_limits<T>::infinity()))
108        maxLog = T(0);  // all -inf (empty region): avoid (-inf) - (-inf) = NaN
109      for (T& e : row)
110        e = std::exp(e - maxLog);
111    }

113    /// Max-shift every row of a freshly computed (hence contiguous)
114    /// log-weight matrix into plain weights, in parallel.
115    template <typename T>

125    /// One weight-matrix entry: the log-weight when the distribution offers
126    /// the channel (rows are then finished by exponentiate_log_rows), else
127    /// the weight itself.

129    T weight_entry(const DistF& distribution, T value)
130    {
131      if constexpr (has_log_weight<DistF, T>)
132        return distribution.log_weight(value);
