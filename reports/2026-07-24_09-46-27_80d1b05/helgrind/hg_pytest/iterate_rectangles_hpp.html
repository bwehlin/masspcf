103    }
104  }
105
106  template <typename PointBiDirIterator, typename FCb>
107  inline void iterate_segments(PointBiDirIterator beginPoints, PointBiDirIterator endPoints,
108    typename std::iterator_traits<PointBiDirIterator>::value_type::time_type a,
109    typename std::iterator_traits<PointBiDirIterator>::value_type::time_type b,
110    FCb cb)
111  {
112    using TPoint = typename std::iterator_traits<PointBiDirIterator>::value_type;
113    using TTime = typename TPoint::time_type;
114    using TVal = typename TPoint::value_type;
115
116    if (beginPoints == endPoints)
117    {
118      return;
119    }
120
121    // float32_t (0 * inf) does not produce 0, so cap b at max instead of using infinity
122    b = std::min(b, (std::numeric_limits<TTime>::max)());
123
124    // Find k = max{i : ti <= a}
125    auto prev = max_time_iterator_prior_to(beginPoints, endPoints, a);
126    auto it = std::next(prev);
127
128    TTime tprev = a;
129    TVal vprev = prev->v;
130
131    Segment<TTime, TVal> seg;
132    for (; it != endPoints; ++it)
133    {
134      TTime tnext = std::min(it->t, b);
135      seg.left = tprev;
136      seg.right = tnext;
137      seg.value = vprev;
138
139      cb(seg);
140
141      if (tnext >= b)
142      {
143        return;
144      }
145
146      tprev = it->t;
147      vprev = it->v;
148    }
149
150    seg.left = tprev;
151    seg.right = b;
152    seg.value = vprev;
153
154    cb(seg);
155  }
