196  // source buffer (PointCloud / DistanceMatrix): each distinct source is
197  // stored once (deduplicated by buffer address — elements sharing a source,
198  // e.g. the indexed subsamples from stablebear.sampling, are written once),
199  // then every element as its source id plus, for indexed views, its index
200  // array. @p sourceKey maps an element to its source buffer address;
201  // @p writeSource writes one element's source.

205  {

215    {

218        idOf.emplace(sourceKey(data[k]), static_cast<uint64_t>(sources.size()));
219        sources.push_back(&data[k]);
220      }
221    }
222

224    write_bytes<uint64_t>(os, static_cast<uint64_t>(sources.size()));
225    for (const ElemT* src : sources)
226    {

228    }

230    // ...then every element as a reference to its source.

309  void write_tensor(std::ostream& os, const TensorT& tensor)
310  {

312    {
313      auto copy = tensor.copy();

327  template <typename T>
328  Tensor<T> read_tensor(std::istream& is)
329  {
330    auto shapeSz = read_bytes<std::uint64_t>(is);
331    std::vector<size_t> shape(shapeSz);
332    std::vector<ptrdiff_t> strides(shapeSz);
333    for (auto i = 0_uz; i < shapeSz; ++i)
334    {
335      shape[i] = read_bytes<std::uint64_t>(is);
336      strides[i] = static_cast<ptrdiff_t>(read_bytes<std::uint64_t>(is));
337    }
338
339    Tensor<T> ret(shape);
340    if (ret.strides() != strides)
341    {
342      throw std::runtime_error("Incorrect strides in saved data (expected " + index_to_string(ret.strides()) + " but got " + index_to_string(strides) + ")");
343    }
344
345    auto sz = ret.size();
346    for (auto * elem = ret.data(); elem != ret.data() + sz; ++elem)
347    {
348      if constexpr (is_barcode_v<T>)
349        *elem = read_barcode<typename is_barcode<T>::scalar_type>(is);
350      else if constexpr (is_compressed_matrix_v<T>)
351        *elem = read_compressed_matrix<T>(is);
352      else if constexpr (is_point_cloud_v<T>)
353        // Legacy (baseFormat 1000) point cloud tensors: every element is a full
354        // nested coordinate tensor.
355        *elem = T(read_element<Tensor<typename is_point_cloud<T>::scalar_type>>(is));
356      else
357        *elem = read_element<T>(is);
358    }
359
360    return ret;
361  }

363  // Shared reader for the shared-source tensor formats (see
364  // write_shared_source_elements): distinct sources stored once, then
365  // per-element (source id, indexed flag, optional indices). Elements that
366  // reference the same source share its buffer, as before saving.
367  // @p readSource reads one source of type SourceT; elements are built as
368  // ElemT(source) or ElemT(source, indices).
369  template <typename ElemT, typename SourceT, typename ReadSourceF>
370  Tensor<ElemT> read_shared_source_tensor(std::istream& is, ReadSourceF readSource)
371  {
372    auto shapeSz = read_bytes<std::uint64_t>(is);
373    std::vector<size_t> shape(shapeSz);
374    std::vector<ptrdiff_t> strides(shapeSz);
375    for (auto i = 0_uz; i < shapeSz; ++i)
376    {
377      shape[i] = read_bytes<std::uint64_t>(is);
378      strides[i] = static_cast<ptrdiff_t>(read_bytes<std::uint64_t>(is));
379    }
380
381    Tensor<ElemT> ret(shape);
382    if (ret.strides() != strides)
383    {
384      throw std::runtime_error("Incorrect strides in saved data (expected " + index_to_string(ret.strides()) + " but got " + index_to_string(strides) + ")");
385    }
386
387    auto numSources = read_bytes<std::uint64_t>(is);
388    std::vector<SourceT> sources;
389    sources.reserve(numSources);
390    for (auto i = 0_uz; i < numSources; ++i)
391    {
392      sources.push_back(readSource(is));
393    }
394
395    auto sz = ret.size();
396    for (auto* elem = ret.data(); elem != ret.data() + sz; ++elem)
397    {
398      auto id = read_bytes<std::uint64_t>(is);
399      const bool indexed = read_bytes<bool>(is);
400      if (indexed)
401      {
402        *elem = ElemT(sources[id], read_element<Tensor<uint64_t>>(is));
403      }
404      else
405      {
406        // Sharing, not copying: PointCloud wraps the coordinate tensor,
407        // DistanceMatrix's copy shares the source buffer (shared_ptr).
408        *elem = ElemT(sources[id]);
409      }
410    }
411
412    return ret;
413  }

451  /// Read a tensor body for element type T, routing on the format already read
452  /// from the stream: the shared-source layout for the current point-cloud and
453  /// distance-matrix formats, element-wise for legacy and unchanged formats.
454  /// Both read entry points go through here so they cannot drift apart.
455  template <typename T>
456  Tensor<T> read_tensor_for_format(std::istream& is, TensorFormat format)
457  {
458    if constexpr (is_point_cloud_v<T>)
459    {
460      if (format == tensorFormat<T>())
461      {
462        return read_indexed_point_cloud_tensor<typename is_point_cloud<T>::scalar_type>(is);
463      }
464    }
465    else if constexpr (is_distance_matrix_v<T>)
466    {
467      if (format == tensorFormat<T>())
468      {
469        return read_indexed_distance_matrix_tensor<typename is_distance_matrix<T>::scalar_type>(is);
470      }
471    }
472    return read_tensor<T>(is);
473  }
