17  /// A point cloud of shape (n_points, dim).
18  ///
19  /// A PointCloud either owns its coordinates or is an indexed view: it shares
20  /// another cloud's coordinate buffer and selects rows through an attached
21  /// index set. Access via n_points()/dim()/operator()(i, j) is transparent to
22  /// which mode it is in, so consumers need no special case. This
23  /// lets a tensor of subsamples store one shared source plus small index arrays
24  /// instead of re-storing every (possibly high-dimensional) point.
25  ///
26  /// Deliberately not a Tensor<T>: the raw coordinate storage and the selected

112        if (keepSource)
113        {
114          return PointCloud(m_coords, m_indices.copy());
115        }
116        return PointCloud(m_coords.copy(), m_indices.copy());
117      }
118      return PointCloud(m_coords.copy());
119    }
120
121    /// Materialize the selected points into a contiguous coordinate tensor.
122    /// Returns the coordinates as-is when not indexed.
123    [[nodiscard]] Tensor<T> materialize() const
124    {
125      if (!is_indexed())
126      {
127        return m_coords;
128      }
129
130      const size_t n = n_points();
131      const size_t d = dim();
132      Tensor<T> out({n, d});
133      for (size_t i = 0; i < n; ++i)
134      {
135        const auto row = static_cast<size_t>(m_indices(i));
136        for (size_t j = 0; j < d; ++j)
137        {
138          out({i, j}) = m_coords({row, j});
139        }
140      }
141      return out;
142    }
143
144  private:
145    Tensor<T> m_coords;         // (n_source_points, dim), possibly shared
146    Tensor<uint64_t> m_indices; // rank-1 when an indexed view, empty otherwise
147  };
148
149  /// Identifies PointCloud<T> instantiations (exposing scalar_type = T), for
150  /// the io and Python binding layers.
151  template <typename T>
152  struct is_point_cloud : std::false_type {};
153
154  template <typename T>
155  struct is_point_cloud<PointCloud<T>> : std::true_type { using scalar_type = T; };
156
157  template <typename T>
158  inline constexpr bool is_point_cloud_v = is_point_cloud<T>::value;

160  /**
161   * Cast a tensor of point clouds (Tensor<PointCloud<U>>) to a different precision
162   * (Tensor<PointCloud<T>>), converting each point cloud's coordinates.
163   *
164   * Sharing carries over: each distinct coordinate buffer is cast once
165   * (deduplicated by buffer address, as in the io layer) and indexed views are
166   * rebuilt on top of the cast source with their own index arrays. Casting cell
167   * by cell through materialize() would instead expand every view into a full
168   * private copy of its source -- exactly the storage blow-up that indexed
169   * subsample tensors exist to avoid.
170   */
171  template <typename T, typename U>
172  requires std::is_constructible_v<T, U>
173  [[nodiscard]] Tensor<PointCloud<T>> pcloud_cast(const Tensor<PointCloud<U>>& src)
174  {
175    Tensor<PointCloud<T>> result(src.shape());
176
177    // Cast each distinct source buffer once...
178    std::unordered_map<const U*, Tensor<T>> castSources;
179    walk(src, [&](const std::vector<size_t>& idx) {
180      const Tensor<U>& coords = src(idx).coords();
181      if (!castSources.contains(coords.data()))
182      {
183        castSources.emplace(coords.data(), tensor_cast<T>(coords));
184      }
185    });
186
187    // ...then rebuild every cell on its shared cast source.
188    walk(src, [&](const std::vector<size_t>& idx) {
189      const PointCloud<U>& cloud = src(idx);
190      const Tensor<T>& source = castSources.at(cloud.coords().data());
191      if (cloud.is_indexed())
192      {
193        result(idx) = PointCloud<T>(source, cloud.indices().copy());
194      }
195      else
196      {
197        result(idx) = PointCloud<T>(source);
198      }
199    });
200    return result;
201  }
