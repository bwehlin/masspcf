37 """Resolve a distance-matrix reference to a single :class:`DistanceMatrix`."""

49def _reject_ambiguous_square_reference(reference_cloud): 

50 """Raise if an implicit point-cloud reference is a square (n, n) array. 

51 

52 A square array is ambiguous: n points in n dimensions, or a precomputed 

53 distance matrix over n points. Reading a distance matrix as a point cloud 

54 silently computes something different (Euclidean distances *between the 

55 matrix rows*) at O(n^3) cost, so square references must declare their 

56 meaning via one of the two wrappers named in the error. 

57 """ 

58 n_points, dim = reference_cloud.shape 

59 if n_points != dim: 

60 return 

61 raise ValueError( 

62 f"reference is a square ({n_points}, {n_points}) array, which is " 

63 "ambiguous: it could be a point cloud or a precomputed distance " 

64 "matrix. Wrap it in stablebear.DistanceMatrix.from_dense(...) to " 

65 "subsample by the stored distances, or in stablebear.FloatTensor(...) " 

66 "to subsample its rows as points." 

67 )

70def _query_is_indices(query): 

71 """Whether ``query`` is a 1-D integer array of reference indices (vs 2-D 

72 coordinates). ``None`` (all reference points) is not indices.""" 

73 if query is None: 

74 return False 

75 arr = np.asarray(query) 

76 return arr.ndim == 1 and np.issubdtype(arr.dtype, np.integer)

79def _reference_indices(query, n): 

80 """Validate a 1-D ``query`` as reference row indices (``None`` means all 

81 ``n`` rows; negative indices count from the end, as in NumPy) and return 

82 a uint64 ndarray.""" 

83 if query is None: 

84 return np.arange(n, dtype=np.uint64) 

85 arr = np.asarray(query).ravel() 

86 # astype copies, so the in-place shift below cannot touch the caller's array. 

87 q = arr.astype(np.int64) 

88 if q.size == 0: 

89 raise ValueError("query must contain at least one index.") 

90 # Only signed input gets the negative-index treatment. A huge unsigned 

91 # value comes out negative from the cast (e.g. uint64 underflow); it must 

92 # fail the range check below, as in NumPy — not wrap to a valid row. 

93 if np.issubdtype(arr.dtype, np.signedinteger): 

94 q[q < 0] += n 

95 if np.any((q < 0) | (q >= n)): 

96 raise ValueError( 

97 f"query indices must be valid reference rows (-{n} <= index < {n}, " 

98 "negative indices count from the end)." 

99 ) 

100 return np.ascontiguousarray(q, dtype=np.uint64)

110def _warn_empty_queries(result): 

111 """Warn once if any query's region held no eligible reference points. 

112 

113 Callers gate this on ``verbose`` — quiet runs proceed silently with the 

114 empty subsamples. Emptiness is a per-query property (an all-zero weight 

115 row empties every instance of that query), so checking one instance per 

116 query suffices. 

117 """ 

118 empty = [q for q in range(result.shape[0]) 

119 if _element_is_empty(result[q, 0])] 

120 if empty: 

121 shown = ", ".join(map(str, empty[:10])) + (", ..." if len(empty) > 10 else "") 

122 warnings.warn( 

123 f"{len(empty)} query point(s) had no reference points with positive " 

124 f"sampling weight; their subsamples are empty (query indices: {shown}).", 

125 stacklevel=4, # helper -> path helper -> subsample_relative -> user code 

126 ) 

127 return result

130def _subsample_distmat(reference, query, *, sample_size, n_instances, distribution, 

131 replace, generator, verbose): 

132 """Distance-matrix path of :func:`subsample_relative` (see its docstring).""" 

133 source = _as_distance_matrix(reference) 

134 if query is not None and not _query_is_indices(query): 

135 raise ValueError( 

136 "a distance matrix has no coordinates; query must be a 1-D integer " 

137 "array of reference row indices (or None for all rows)." 

138 ) 

139 query_indices = IntTensor(_reference_indices(query, source.size), dtype=uint64) 

140 backend = _backend(source.dtype) 

141 

142 # Fully fused C++ draw: precomputed distances + built-in distribution. 

143 task, result = backend.sample_subsets_distmat( 

144 source._data, query_indices._data, distribution._descriptor(backend), 

145 sample_size, n_instances, replace, _unwrap(generator) 

146 ) 

147 _run_task(lambda: task, verbose=verbose) 

148 

149 if verbose: 

150 return _warn_empty_queries(DistanceMatrixTensor(result)) 

151 return DistanceMatrixTensor(result)

154def _subsample_pointcloud(reference, query, *, sample_size, n_instances, distribution, 

155 replace, generator, verbose): 

156 """Point-cloud path of :func:`subsample_relative` (see its docstring).""" 

157 # An explicit FloatTensor declares "rows are points"; only raw arrays are 

158 # screened by the square-array ambiguity guard. 

159 explicit_cloud = isinstance(reference, FloatTensor) 

160 reference_cloud = _as_float_tensor(reference) 

161 if reference_cloud.ndim != 2: 

162 raise ValueError( 

163 "reference must be a 2-D (n_reference, dim) point cloud, but got " 

164 f"{reference_cloud.ndim} dimension(s)." 

165 ) 

166 if not explicit_cloud: 

167 _reject_ambiguous_square_reference(reference_cloud) 

168 if query is None: 

169 query_cloud = reference_cloud 

170 elif _query_is_indices(query): 

171 # Query points selected by their order in the reference cloud. 

172 row_indices = _reference_indices(query, reference_cloud.shape[0]) 

173 query_cloud = _as_float_tensor( 

174 np.asarray(reference_cloud)[row_indices], dtype=reference_cloud.dtype 

175 ) 

176 else: 

177 query_cloud = _as_float_tensor(query, dtype=reference_cloud.dtype) 

178 if query_cloud.ndim != 2: 

179 raise ValueError( 

180 "query must be a 2-D (n_query, dim) array of coordinates or a 1-D " 

181 "integer array of reference indices." 

182 ) 

183 

184 if reference_cloud.shape[1] != query_cloud.shape[1]: 

185 raise ValueError("reference and query must have the same dimension.") 

186 

187 backend = _backend(reference_cloud.dtype) 

188 

189 # Fully fused C++ draw: distances + built-in distribution. 

190 task, result = backend.sample_subsets( 

191 reference_cloud._data, query_cloud._data, backend.Euclidean(), 

192 distribution._descriptor(backend), sample_size, n_instances, replace, 

193 _unwrap(generator) 

194 ) 

195 _run_task(lambda: task, verbose=verbose) 

196 if verbose: 

197 return _warn_empty_queries(PointCloudTensor(result)) 

198 return PointCloudTensor(result)

201def subsample_relative( 

202 reference, 

203 query=None, 

204 *, 

205 sample_size, 

206 n_instances, 

207 distribution=None, 

208 replace=True, 

209 generator=None, 

210 verbose=False, 

211): 

212 r"""Subsample a reference point cloud relative to each query point. 

213 

214 Front end of the relative-approach pipeline of Agerberg, Chacholski & 

215 Ramanujam (2023). For each query point :math:`p`, each reference point 

216 :math:`r` is weighted by :math:`D(\lVert p - r\rVert)` for the chosen 

217 distribution :math:`D`, and ``n_instances`` subsamples of up to 

218 ``sample_size`` points are drawn with those weights. See :doc:`the 

219 subsampling guide </sampling>` for background, examples, and figures. 

220 

221 The reference may also be a precomputed distance matrix 

222 (:class:`~stablebear.DistanceMatrix` or a one-element 

223 :class:`~stablebear.DistanceMatrixTensor`): distances are then read from 

224 the matrix, the query points are row indices into it, and each subsample 

225 is the principal sub-distance-matrix over the drawn points. 

226 

227 Parameters 

228 ---------- 

229 reference : array_like, FloatTensor, DistanceMatrix, or DistanceMatrixTensor 

230 The reference point cloud, shape ``(n_reference, dim)``, or a 

231 precomputed distance matrix over the reference points. A raw square 

232 ``(n, n)`` array is ambiguous between the two and is rejected: wrap 

233 it in :class:`~stablebear.DistanceMatrix` (or a one-element 

234 :class:`~stablebear.DistanceMatrixTensor`) to subsample by stored 

235 distances, or in :class:`~stablebear.FloatTensor` to subsample its 

236 rows as points. 

237 query : array_like, optional 

238 A 1-D integer array of reference indices (negative indices count from 

239 the end, as in NumPy), or a 2-D ``(n_query, dim)`` array of 

240 coordinates; coordinates require a point-cloud reference. If ``None`` 

241 (the default), every reference point is used as a query point. 

242 sample_size : int 

243 Maximum number of points per subsample. A subsample is smaller only 

244 when it runs out of eligible points: drawing without replacement with 

245 fewer than ``sample_size`` positively weighted reference points, or a 

246 query point with no positively weighted reference points at all 

247 (empty subsamples; reported per query when ``verbose=True``). 

248 n_instances : int 

249 Number of subsamples drawn per query point. 

250 distribution : Gaussian or Uniform, optional 

251 Maps distances to non-negative sampling weights. By default 

252 :class:`Gaussian` with ``mean=0``, ``sigma=1``. 

253 replace : bool, optional 

254 Whether to draw with replacement, by default True (as in the paper). 

255 generator : Generator, optional 

256 Random number generator. If ``None``, the global generator is used. 

257 verbose : bool, optional 

258 If True, display a progress bar, allow cooperative cancellation 

259 (e.g. via ``KeyboardInterrupt``), and warn (:class:`UserWarning`) 

260 about query points whose subsamples came out empty. By default False. 

261 

262 Returns 

263 ------- 

264 PointCloudTensor or DistanceMatrixTensor 

265 Tensor of shape ``(n_query, n_instances)``; element ``[i, j]`` is the 

266 ``j``-th subsample for query point ``i`` — a point cloud of at most 

267 ``sample_size`` points, or the sub-distance-matrix over the drawn 

268 points for distance-matrix input. 

269 """ 

270 sample_size = operator.index(sample_size) 

271 n_instances = operator.index(n_instances) 

272 if sample_size <= 0: 

273 raise ValueError("sample_size must be positive.") 

274 if n_instances <= 0: 

275 raise ValueError("n_instances must be positive.") 

276 if distribution is None: 

277 distribution = Gaussian(mean=0.0, sigma=1.0) 

278 if not isinstance(distribution, _BUILTIN_DISTRIBUTIONS): 

279 raise ValueError( 

280 "distribution must be a built-in distribution " 

281 f"({' or '.join(d.__name__ for d in _BUILTIN_DISTRIBUTIONS)}) or None." 

282 ) 

283 

284 # Distance-matrix input: weight by precomputed distances, return sub-matrices. 

285 if isinstance(reference, (DistanceMatrix, DistanceMatrixTensor)): 

286 return _subsample_distmat( 

287 reference, query, sample_size=sample_size, n_instances=n_instances, 

288 distribution=distribution, replace=replace, generator=generator, verbose=verbose, 

289 ) 

290 # Point-cloud input: weight by Euclidean distances computed from coordinates. 

291 return _subsample_pointcloud( 

292 reference, query, sample_size=sample_size, n_instances=n_instances, 

293 distribution=distribution, replace=replace, generator=generator, verbose=verbose, 

294 )
