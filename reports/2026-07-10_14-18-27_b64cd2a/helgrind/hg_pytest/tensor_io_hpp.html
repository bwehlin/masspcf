320  void write_tensor(std::ostream& os, const TensorT& tensor)
321  {
322    if (!tensor.is_contiguous())
323    {
324      auto copy = tensor.copy();
325      if (!copy.is_contiguous())

338  template <typename T>
339  Tensor<T> read_tensor(std::istream& is)
340  {
341    auto shapeSz = read_bytes<std::uint64_t>(is);
342    std::vector<size_t> shape(shapeSz);
343    std::vector<ptrdiff_t> strides(shapeSz);
344    for (auto i = 0_uz; i < shapeSz; ++i)
345    {
346      shape[i] = read_bytes<std::uint64_t>(is);
347      strides[i] = static_cast<ptrdiff_t>(read_bytes<std::uint64_t>(is));
348    }
349
350    Tensor<T> ret(shape);
351    if (ret.strides() != strides)
352    {
353      throw std::runtime_error("Incorrect strides in saved data (expected " + index_to_string(ret.strides()) + " but got " + index_to_string(strides) + ")");
354    }
355
356    auto sz = ret.size();
357    for (auto * elem = ret.data(); elem != ret.data() + sz; ++elem)
358    {
359      if constexpr (is_barcode_v<T>)
360        *elem = read_barcode<typename is_barcode<T>::scalar_type>(is);
361      else if constexpr (is_compressed_matrix_v<T>)
362        *elem = read_compressed_matrix<T>(is);
363      else
364        *elem = read_element<T>(is);
365    }
366
367    return ret;
368  }

371  // coordinate buffers stored once, then per-element (source id, optional indices).
372  // Elements that reference the same source share its coordinate buffer.
373  template <typename ScalarT>
374  Tensor<PointCloud<ScalarT>> read_indexed_point_cloud_tensor(std::istream& is)
375  {
376    auto shapeSz = read_bytes<std::uint64_t>(is);
377    std::vector<size_t> shape(shapeSz);
378    std::vector<ptrdiff_t> strides(shapeSz);
379    for (auto i = 0_uz; i < shapeSz; ++i)
380    {
381      shape[i] = read_bytes<std::uint64_t>(is);
382      strides[i] = static_cast<ptrdiff_t>(read_bytes<std::uint64_t>(is));
383    }
384
385    Tensor<PointCloud<ScalarT>> ret(shape);
386    if (ret.strides() != strides)
387    {
388      throw std::runtime_error("Incorrect strides in saved data (expected " + index_to_string(ret.strides()) + " but got " + index_to_string(strides) + ")");
389    }
390
391    auto numSources = read_bytes<std::uint64_t>(is);
392    std::vector<Tensor<ScalarT>> sources;
393    sources.reserve(numSources);
394    for (auto i = 0_uz; i < numSources; ++i)
395      sources.push_back(read_element<Tensor<ScalarT>>(is));
396
397    auto sz = ret.size();
398    for (auto * elem = ret.data(); elem != ret.data() + sz; ++elem)
399    {
400      auto id = read_bytes<std::uint64_t>(is);
401      auto indexed = read_bytes<std::uint8_t>(is);
402      if (indexed)
403        *elem = PointCloud<ScalarT>(sources[id], read_element<Tensor<uint64_t>>(is));
404      else
405        *elem = PointCloud<ScalarT>(sources[id]);
406    }
407
408    return ret;
409  }
