44  /// Map a @p target in [0, cdf.back()] to the index whose CDF interval
45  /// contains it. uniform_real_distribution may round up to exactly
46  /// cdf.back() (LWG 2524); such a target is pulled back inside the range,
47  /// so the draw lands on the last positive-weight interval — never past it
48  /// onto a trailing zero-weight (ineligible) point.
49  template <typename T>
50  size_t index_for_target(std::span<const T> cdf, T target)
51  {
52    if (target >= cdf.back())
53      target = std::nextafter(cdf.back(), T(0));
54    const auto it = std::upper_bound(cdf.begin(), cdf.end(), target);
55    return static_cast<size_t>(it - cdf.begin());
56  }
